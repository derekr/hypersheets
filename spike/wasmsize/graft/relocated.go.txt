package main

// rowHeightPx is one row. Copied from render.go's layout geometry block.
const rowHeightPx = 22

// rowRange is an inclusive row range. Hi < Lo means empty. Copied from window.go.
type rowRange struct{ Lo, Hi int }

func (r rowRange) empty() bool { return r.Hi < r.Lo }

func (r rowRange) rows() int {
	if r.empty() {
		return 0
	}
	return r.Hi - r.Lo + 1
}
